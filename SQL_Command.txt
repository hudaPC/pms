SELECT
    i.item_code,
    i.item_name,
    pmi.old_part_no,
    pmi.revision,
    pmi.quantity,
    s.name AS supplier_name,
    isup.unit_price,
    isup.lead_time_days,
    isup.minimum_order_qty
FROM product_models pm
INNER JOIN product_model_items pmi
    ON pmi.product_model_id = pm.id
INNER JOIN items i
    ON i.id = pmi.item_id
LEFT JOIN item_suppliers isup
    ON isup.item_id = i.id
   AND isup.is_preferred = 1
   AND isup.status = 1
LEFT JOIN suppliers s
    ON s.id = isup.supplier_id
WHERE pm.model_code = 'VS-0814H1'
ORDER BY i.item_code;

SELECT
    i.item_code,
    i.item_name,
    pmi.old_part_no,
    pmi.revision,
    pmi.quantity,
    s.name AS supplier_name,
    isup.unit_price,
    isup.lead_time_days,
    isup.minimum_order_qty
FROM product_models pm
INNER JOIN product_model_items pmi
    ON pmi.product_model_id = pm.id
INNER JOIN items i
    ON i.id = pmi.item_id
LEFT JOIN item_suppliers isup
    ON isup.item_id = i.id
   AND isup.status = 1
   AND isup.id = (
        SELECT isup2.id
        FROM item_suppliers isup2
        WHERE isup2.item_id = i.id
          AND isup2.status = 1
        ORDER BY isup2.is_preferred DESC,   -- preferred first
                 isup2.supplier_id = i.default_supplier_id DESC, -- then default supplier
                 isup2.unit_price ASC        -- then cheapest as tiebreaker
        LIMIT 1
   )
LEFT JOIN suppliers s
    ON s.id = isup.supplier_id
WHERE pm.model_code = 'TM-2528H1-850'
ORDER BY i.item_code;
